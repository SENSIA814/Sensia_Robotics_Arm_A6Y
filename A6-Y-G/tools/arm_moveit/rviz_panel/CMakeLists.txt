cmake_minimum_required(VERSION 3.16)
project(a6y_rviz_panel)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(${PROJECT_NAME} SHARED planning_panel.cpp planning_panel.hpp)
ament_target_dependencies(${PROJECT_NAME} rviz_common rclcpp pluginlib std_msgs)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_panel_close test_panel_close.cpp)
  ament_target_dependencies(test_panel_close rviz_common rclcpp std_msgs)
  target_link_libraries(test_panel_close ${PROJECT_NAME} Qt5::Widgets)
  add_test(NAME panel_close COMMAND test_panel_close)
  set_tests_properties(panel_close PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
