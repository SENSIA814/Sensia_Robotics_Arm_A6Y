cmake_minimum_required(VERSION 3.8)
project(a6_y_ros2_control)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(a6_y_damiao_system SHARED src/A6_y_damiao_system.cpp src/damiao_can.cpp)
target_compile_features(a6_y_damiao_system PUBLIC cxx_std_17)
target_include_directories(a6_y_damiao_system PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(a6_y_damiao_system hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface A6_y_ros2_control_plugin.xml)
add_executable(damiao_protocol_check src/damiao_protocol_check.cpp src/damiao_can.cpp)
target_compile_features(damiao_protocol_check PUBLIC cxx_std_17)
target_include_directories(damiao_protocol_check PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
if(BUILD_TESTING)
  enable_testing()
  find_package(PythonInterp 3 REQUIRED)
  add_executable(test_accepted_position test/test_accepted_position.cpp)
  target_include_directories(test_accepted_position PRIVATE include)
  add_test(NAME accepted_position COMMAND test_accepted_position)
  add_test(NAME damiao_protocol_check COMMAND $<TARGET_FILE:damiao_protocol_check>)
  add_test(NAME damiao_feedback_watch_pty COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_damiao_feedback_watch.py)
  set_tests_properties(damiao_feedback_watch_pty PROPERTIES ENVIRONMENT "PYTHONNOUSERSITE=1" TIMEOUT 60)
  add_test(NAME damiao_set_id COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_damiao_set_id.py)
  set_tests_properties(damiao_set_id PROPERTIES ENVIRONMENT "PYTHONNOUSERSITE=1" TIMEOUT 60)
  add_test(NAME motor_params COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_motor_params.py)
  set_tests_properties(motor_params PROPERTIES ENVIRONMENT "PYTHONNOUSERSITE=1" TIMEOUT 60)
  add_test(NAME protocol_vmax COMMAND ${PYTHON_EXECUTABLE} -m pytest -q ${CMAKE_CURRENT_SOURCE_DIR}/test/test_protocol_vmax.py)
  set_tests_properties(protocol_vmax PROPERTIES ENVIRONMENT "PYTHONNOUSERSITE=1" TIMEOUT 30)
endif()
install(TARGETS a6_y_damiao_system EXPORT export_a6_y_damiao_system ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS damiao_protocol_check RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY "Motor Drive" DESTINATION share/${PROJECT_NAME})
install(PROGRAMS tools/damiao_bus_scan.py tools/damiao_feedback_watch.py tools/damiao_set_id.py tools/motor_status.py
  tools/read_motor_params.py tools/write_motor_params_temp.py tools/save_motor_params_flash.py
  DESTINATION lib/${PROJECT_NAME})
install(FILES A6_y_ros2_control_plugin.xml DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_a6_y_damiao_system HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle)
ament_package()

install(PROGRAMS tools/move_to_group_state.py DESTINATION lib/${PROJECT_NAME})
